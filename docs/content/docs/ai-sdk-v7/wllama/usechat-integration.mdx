## Overview

When using this library with the `useChat` hook, you'll need to create a [custom transport](https://ai-sdk.dev/docs/ai-sdk-ui/transport) implementation to handle client-side AI with download progress.
This is not required, but it makes it easier for you to build better user experiences.

wllama runs inference in its own Web Worker, so unlike other providers no manual worker setup is needed.

## Complete Transport Example

```ts
import {
  ChatTransport,
  UIMessageChunk,
  streamText,
  convertToModelMessages,
  ChatRequestOptions,
  createUIMessageStream,
  wrapLanguageModel,
  extractReasoningMiddleware,
  tool,
  isStepCount,
  toUIMessageStream,
} from "ai";
import { WllamaUIMessage, WllamaLanguageModel } from "@browser-ai/wllama";
import z from "zod";

export const createTools = () => ({
  webSearch: tool({
    description:
      "Search the web for information when you need up-to-date information or facts not in your knowledge base. Use this when the user asks about current events, recent developments, or specific factual information you're unsure about.",
    inputSchema: z.object({
      query: z
        .string()
        .describe("The search query to find information on the web"),
    }),
    execute: async ({ query }) => {
      // ...
    },
  }),
});

/**
 * Client-side chat transport AI SDK implementation that handles AI model communication
 * with in-browser AI capabilities (wllama / llama.cpp on WebGPU or CPU).
 *
 * @implements {ChatTransport<WllamaUIMessage>}
 */
export class WllamaChatTransport implements ChatTransport<WllamaUIMessage> {
  private readonly model: WllamaLanguageModel;
  private tools: ReturnType<typeof createTools>;

  constructor(model: WllamaLanguageModel) {
    this.model = model;
    this.tools = createTools();
  }

  async sendMessages(
    options: {
      chatId: string;
      messages: WllamaUIMessage[];
      abortSignal: AbortSignal | undefined;
    } & {
      trigger: "submit-message" | "submit-tool-result" | "regenerate-message";
      messageId: string | undefined;
    } & ChatRequestOptions,
  ): Promise<ReadableStream<UIMessageChunk>> {
    const { messages, abortSignal } = options;
    const prompt = await convertToModelMessages(messages);
    const model = this.model;

    return createUIMessageStream<WllamaUIMessage>({
      execute: async ({ writer }) => {
        let downloadProgressId: string | undefined;
        const availability = await model.availability();

        // Only track progress if model needs downloading
        if (availability !== "available") {
          await model.createSessionWithProgress((progress) => {
            const percent = Math.round(progress * 100);

            if (progress >= 1) {
              if (downloadProgressId) {
                writer.write({
                  type: "data-modelDownloadProgress",
                  id: downloadProgressId,
                  data: {
                    status: "complete",
                    progress: 100,
                    message:
                      "Model finished downloading! Getting ready for inference...",
                  },
                });
              }
              return;
            }

            if (!downloadProgressId) {
              downloadProgressId = `download-${Date.now()}`;
            }

            writer.write({
              type: "data-modelDownloadProgress",
              id: downloadProgressId,
              data: {
                status: "downloading",
                progress: percent,
                message: `Downloading browser AI model... ${percent}%`,
              },
              transient: !downloadProgressId,
            });
          });
        }

        const result = streamText({
          // Surface <think> blocks as reasoning parts instead of inline text
          model: wrapLanguageModel({
            model,
            middleware: extractReasoningMiddleware({ tagName: "think" }),
          }),
          tools: this.tools,
          stopWhen: isStepCount(5),
          messages: prompt,
          abortSignal,
          onChunk: (event) => {
            if (event.chunk.type === "text-delta" && downloadProgressId) {
              writer.write({
                type: "data-modelDownloadProgress",
                id: downloadProgressId,
                data: { status: "complete", progress: 100, message: "" },
              });
              downloadProgressId = undefined;
            }
          },
        });

        writer.merge(
          toUIMessageStream({
            stream: result.stream,
            tools: this.tools,
            sendStart: false,
          }),
        );
      },
    });
  }

  async reconnectToStream(
    options: {
      chatId: string;
    } & ChatRequestOptions,
  ): Promise<ReadableStream<UIMessageChunk> | null> {
    // Client-side AI doesn't support stream reconnection
    return null;
  }
}
```

## Basic Transport Structure

Here's a simplified example of how to structure a custom transport:

```typescript
import {
  ChatTransport,
  UIMessageChunk,
  streamText,
  convertToModelMessages,
  ChatRequestOptions,
  toUIMessageStream,
} from "ai";
import { WllamaLanguageModel, WllamaUIMessage } from "@browser-ai/wllama";

// This class won't stream back data parts with the download progress if
// the model hasn't yet been downloaded
export class SimpleWllamaChatTransport implements ChatTransport<WllamaUIMessage> {
  private readonly model: WllamaLanguageModel;

  constructor(model: WllamaLanguageModel) {
    this.model = model;
  }

  async sendMessages(
    options: {
      chatId: string;
      messages: WllamaUIMessage[];
      abortSignal: AbortSignal | undefined;
    } & {
      trigger: "submit-message" | "submit-tool-result" | "regenerate-message";
      messageId: string | undefined;
    } & ChatRequestOptions,
  ): Promise<ReadableStream<UIMessageChunk>> {
    const prompt = convertToModelMessages(options.messages);

    const result = streamText({
      model: this.model,
      messages: prompt,
      abortSignal: options.abortSignal,
    });

    return toUIMessageStream({ stream: result.stream });
  }

  async reconnectToStream(
    options: {
      chatId: string;
    } & ChatRequestOptions,
  ): Promise<ReadableStream<UIMessageChunk> | null> {
    // Client-side AI doesn't support stream reconnection
    return null;
  }
}
```

You can then use the `useChat()` hook in your components:

```ts
import { wllama, WllamaUIMessage } from "@browser-ai/wllama";

const model = wllama("hf:ggml-org/Qwen3-0.6B-GGUF/Qwen3-0.6B-Q8_0.gguf", {
  loadParams: { n_ctx: 4096 },
});

const { sendMessage, messages, stop } = useChat<WllamaUIMessage>({
  transport: new WllamaChatTransport(model),
});
```

In case the device is incompatible with local in-browser LLMs, and we want to use a server-side AI model, we can simply use the utility function from the package first:

```ts
import {
  wllama,
  WllamaUIMessage,
  doesBrowserSupportWllama,
} from "@browser-ai/wllama";

const model = wllama("hf:ggml-org/Qwen3-0.6B-GGUF/Qwen3-0.6B-Q8_0.gguf", {
  loadParams: { n_ctx: 4096 },
});

const { sendMessage, messages, stop } = useChat<WllamaUIMessage>({
  transport: doesBrowserSupportWllama() // check for device compatibility
    ? new WllamaChatTransport(model)
    : new DefaultChatTransport<UIMessage>({
        api: "/api/chat",
      }),
});
```

<Callout type="info">
  wllama only requires WebAssembly and falls back to CPU inference when
  WebGPU is unavailable, so `doesBrowserSupportWllama()` is `true` in almost
  every browser. The server fallback above is mainly relevant for very old
  browsers without WebAssembly support.
</Callout>
